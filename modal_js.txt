var amRev=null;
function openApplyModal(){setTimeout(function(){var el=document.getElementById('applyModal');if(el){el.classList.add('open');document.body.style.overflow='hidden';}var bar=document.getElementById('amBar');if(bar)bar.style.width='0%';['amS1','amS2','amS3','amGo','amNo'].forEach(function(id){var s=document.getElementById(id);if(s)s.classList.remove('active')});var s1=document.getElementById('amS1');if(s1)s1.classList.add('active');document.querySelectorAll('.am-opt').forEach(function(b){b.classList.remove('sel')});amRev=null;},20);}
function closeApplyModal(){var el=document.getElementById('applyModal');if(el)el.classList.remove('open');document.body.style.overflow='';}
function amPick(btn,step){btn.closest('.am-step').querySelectorAll('.am-opt').forEach(function(b){b.classList.remove('sel')});btn.classList.add('sel');var v=btn.getAttribute('data-v');if(step===1){amRev=v;setTimeout(function(){amShow('amS2');var bar=document.getElementById('amBar');if(bar)bar.style.width='33%';},350);}else if(step===2){setTimeout(function(){amShow('amS3');var bar=document.getElementById('amBar');if(bar)bar.style.width='66%';},350);}else if(step===3){var bar=document.getElementById('amBar');if(bar)bar.style.width='100%';setTimeout(function(){amShow((amRev!=='low'&&v==='yes')?'amGo':'amNo');},350);}}
function amShow(id){['amS1','amS2','amS3','amGo','amNo'].forEach(function(i){var el=document.getElementById(i);if(el)el.classList.remove('active')});var t=document.getElementById(id);if(t)t.classList.add('active');}
document.addEventListener('keydown',function(e){if(e.key==='Escape')closeApplyModal();});
document.addEventListener('click',function(e){if(e.target&&e.target.id==='applyModal')closeApplyModal();});
document.querySelectorAll('.ins-inner').forEach(function(el){var d=document.createElement('div');d.className='ins-art-cta';d.innerHTML='<p>If this resonates with where your business is right now, this is exactly what we fix.</p><a href="javascript:void(0)" onclick="openApplyModal()"><button class="btn bp" style="font-size:11px;padding:10px 26px;margin-top:8px">Apply to Work With Kryson →</button></a>';el.appendChild(d);});
